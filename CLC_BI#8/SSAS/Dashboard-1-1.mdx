-- 1. So sánh OTP và Thời gian trễ (Khởi hành/Đến) trung bình theo hãng

WITH 
-- Tính OTP Rate (Sử dụng Measure tính toán)
MEMBER [Measures].[OTP %] AS
    IIF([Measures].[Fact Flights Count] = 0, 0, [Measures].[On Time Count] / [Measures].[Fact Flights Count]), 
    FORMAT_STRING = "Percent"

-- Tính Trễ khởi hành TB (Sử dụng Measure tính toán)
MEMBER [Measures].[Avg Dep Delay] AS
    [Measures].[Dep Delay] / [Measures].[Fact Flights Count], 
    FORMAT_STRING = "Standard"

-- Tính Trễ đến TB (Sử dụng Measure tính toán)
MEMBER [Measures].[Avg Arr Delay] AS
    [Measures].[Arr Delay] / [Measures].[Fact Flights Count], 
    FORMAT_STRING = "Standard"

SELECT 
    { [Measures].[OTP %], [Measures].[Avg Dep Delay], [Measures].[Avg Arrival Delay] } ON COLUMNS,
    NON EMPTY { [Dim Airline].[Airline ID].MEMBERS } ON ROWS
FROM [DDS Airline]

-- ============================================================================
-- 1. FLIGHT VOLUME ANALYSIS BY TIME PERIODS
-- ============================================================================
-- Description: Analyze total flight counts aggregated by different time granularities
-- Business Use: Trend analysis, capacity planning, seasonal pattern identification
-- ============================================================================

-- 1.1 Annual Flight Count
-- Returns: Total flights per calendar year
SELECT 
    {[Measures].[Flights Count]} ON COLUMNS,
    [Dim Date].[Calendar].[Year].Members ON ROWS
FROM [DDS Airline];

-- 1.2 Quarterly Flight Count
-- Returns: Total flights per quarter for trend analysis
SELECT 
    {[Measures].[Flights Count]} ON COLUMNS,
    [Dim Date].[Calendar].[Quarter].Members ON ROWS
FROM [DDS Airline];

-- 1.3 Monthly Flight Count
-- Returns: Total flights per month for detailed seasonality analysis
SELECT 
    {[Measures].[Flights Count]} ON COLUMNS,
    [Dim Date].[Calendar].[Month Num].Members ON ROWS
FROM [DDS Airline];


-- ============================================================================
-- 2. AIRPORT TRAFFIC ANALYSIS
-- ============================================================================
-- Description: Identify busiest airports by total flight operations
-- Business Use: Resource allocation, infrastructure planning, hub identification
-- ============================================================================

-- 2.1 Top 5 Busiest Origin Airports
-- Returns: Airports with highest departure flight volumes
-- Note: Ranked by total departures; consider adding destination airport analysis
SELECT 
    {[Measures].[Flights Count]} ON COLUMNS,
    TOPCOUNT(
        [Origin Airport].[Airport ID].Members, 
        5, 
        [Measures].[Flights Count]
    ) ON ROWS
FROM [DDS Airline];


-- ============================================================================
-- 3. ON-TIME PERFORMANCE (OTP) METRICS
-- ============================================================================
-- Description: Calculate on-time performance ratio by origin airport
-- Business Use: Service quality monitoring, SLA compliance, airport efficiency
-- OTP Definition: Flights arriving/departing within ±5 minutes of scheduled time
-- ============================================================================

-- 3.1 OTP Ratio by Origin Airport
-- Returns: Percentage of on-time flights per airport
-- Formula: (On-Time Flights / Total Flights) × 100
-- Excludes: Airports with zero flights to prevent division errors
WITH MEMBER [Measures].[OTP %] AS
    IIF(
        [Measures].[Flights Count] = 0, 
        NULL,
        ([Measures].[On Time Count] / [Measures].[Flights Count])
    ), 
    FORMAT_STRING = "Percent"

SELECT 
    {[Measures].[OTP %]} ON COLUMNS,
    NON EMPTY [Origin Airport].[Airport ID].Members ON ROWS
FROM [DDS Airline];


-- ============================================================================
-- 4. FLIGHT CANCELLATION ANALYSIS
-- ============================================================================
-- Description: Analyze cancellation patterns by root cause
-- Business Use: Operational improvement, cost reduction, risk mitigation
-- ============================================================================

-- 4.1 Cancellation Rate by Reason
-- Returns: Percentage of cancelled flights categorized by cancellation reason
-- Formula: (Cancelled Flights by Reason / Total Flights) × 100
-- Common Reasons: Weather, Airline Operations, Security, Air Traffic Control
WITH MEMBER [Measures].[Cancellation Rate] AS
    IIF(
        [Measures].[Flights Count] = 0,
        NULL,
        ([Measures].[Sum Cancelled] / [Measures].[Flights Count])
    ),
    FORMAT_STRING = "Percent"

SELECT
    {[Measures].[Cancellation Rate]} ON COLUMNS,
    [Dim Reason].[Reason ID].Members ON ROWS
FROM [DDS Airline];


-- ============================================================================
-- 5. DELAY PERFORMANCE METRICS
-- ============================================================================
-- Description: Calculate average delay duration by airport
-- Business Use: Delay pattern identification, operational efficiency improvement
-- Metric Unit: Minutes
-- ============================================================================

-- 5.1 Average Departure Delay by Origin Airport
-- Returns: Mean departure delay time (minutes) per origin airport
-- Positive values: Delayed departures | Negative values: Early departures
SELECT
    {[Measures].[Avg Dep Delay]} ON COLUMNS,
    NON EMPTY [Origin Airport].[Airport ID].Members ON ROWS
FROM [DDS Airline];

-- 5.2 Average Arrival Delay by Destination Airport
-- Returns: Mean arrival delay time (minutes) per destination airport
-- Positive values: Delayed arrivals | Negative values: Early arrivals
SELECT
    {[Measures].[Avg Arrival Delay]} ON COLUMNS,
    NON EMPTY [Dest Airport].[Airport ID].Members ON ROWS
FROM [DDS Airline];
